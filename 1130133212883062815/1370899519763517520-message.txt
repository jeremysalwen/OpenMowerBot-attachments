mqtt:
  sensor:

  - state_topic: "robot_state/json"
    name: "OpenMower Position Lat"
    unique_id: openmower_position_lat
    icon: mdi:ruler
    value_template: >-
      {# https://stackoverflow.com/a/50506609 #}
      {# https://github.com/Turfjs/turf/issues/635#issuecomment-292011500 #}
      {% set earth = 6371008.8 %}
      {% set m = 1 / ((2 * pi / 360) * earth) %}
      {% set datum_lat = xx.xxxxx %} # enter your mower base latitude here
      {% set datum_lon = xx.xxxxx %} # enter your mower base longitude here
      {{ datum_lat + (value_json.pose.y * m) }}


  - state_topic: "robot_state/json"
    name: "OpenMower Position Lon"
    unique_id: openmower_position_lon
    icon: mdi:ruler
    value_template: >-
      {# https://stackoverflow.com/a/50506609 #}
      {# https://github.com/Turfjs/turf/issues/635#issuecomment-292011500 #}
      {% set earth = 6371008.8 %}
      {% set m = 1 / ((2 * pi / 360) * earth) %}
      {% set datum_lat = xx.xxxxx %} # enter your mower base latitude here
      {% set datum_lon = xx.xxxxx %} # enter your mower base longitude here
      {{ datum_lon + (value_json.pose.x * m) / cos(datum_lat * (pi / 180)) }}
